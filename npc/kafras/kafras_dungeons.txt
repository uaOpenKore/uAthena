// Mocfild04 -----------------------------------------
moc_fild04,230,329,3	script	Kafra::kaf_moc_fild04	115,{

    cutin "kafra_01",2;
    cleararray @wrpC$[0],"",getarraysize(@wrpC$);
    setarray @wrpD$[0], "Prontera", "Morroc";
    setarray @wrpP[0], 1200, 1200;
set @wrpC$[0], @wrpD$[0]+" -> "+@wrpP[0];
set @wrpC$[1], @wrpD$[1]+" -> "+@wrpP[1];
set @wrpC$[2], "Cancel";
callfunc "F_Kafra",0,0,1;

	M_Save:
		savepoint "moc_fild04",231,339;
		callfunc "F_KafEnd",0,1;
}

// Glastheim ----------------------------------------------------------------
glast_01,202,82,4	script	Kafra	115,{
    mes "[Kafra]";
    mes "Welcome to Kafra Service!";
    mes "Would you like to go back to the world of the living?";
    mes "Teleport to Prontera costs ^FF00002,000 Zeny^000000.";
    next;

    switch(select("Teleport to Prontera:Open Storage:Cancel")) {
        case 1:
            if (Zeny < 2000) {
                mes "[Kafra]";
                mes "I'm sorry, but you don't have enough Zeny.";
                close;
            }   
            set Zeny, Zeny - 2000;
            specialeffect2 EF_PORTAL1;
            warp "prontera",116,72;
            end;
                
        case 2:
            close2;
            openstorage;
            end;

        case 3:
            mes "[Kafra]";
            mes "Have a safe journey!";
            close;
    }
}
