
hugel,88,168,4	script	Cool Event Corp. Staff#hugel	874,{

	mes "[Cool Event Corp. Staff]";
	if(#gl_lang == 1) {
		mes "Ласкаво просимо до Cool Event Corp.";
		mes "Наш персонал завжди працює над тим, щоб перевершити ваші очікування щодо якості сервісу. Тож чим я можу допомогти вам сьогодні?";
	} else {
		mes "Welcome to Cool Event Corp.";
		mes "Our staff is always working to surpass your expactations for quality service. So how may I assist you today?";
	}
	next;

	switch(select("Save:Use Storage:Rent a Pushcart:Cancel")){
	case 1:
		mes "[Cool Event Corp. Staff]";
		if(#gl_lang == 1) {
			mes "Вашу точку повернення було збережено тут, у селищі Х'югель.";
			mes "Дякуємо, що скористалися послугами Cool Event Corp.~";
		} else {
			mes "Your Respawn Point has been saved here in the village of Hugel.";
			mes "Thank you for using the Cool Event Corp. service~";
		}
		emotion e_thx;
		savepoint "hugel",96,155;
		close;
	case 2:
		mes "[Cool Event Corp. Staff]";
		if(basicskillcheck() && getskilllv(1) < 6){
			if(#gl_lang == 1) mes "Мені шкода, але вам потрібен 6 рівень базових навичок Новачка для використання складу.";
			else mes "I'm sorry, but you need the Novice's Basic Skill Level 6 to use the Storage Service.";
			emotion e_sry;
			close;
		}
		if(Zeny < 40){
			if(#gl_lang == 1) mes "Вибачте, у вас недостатньо Зені. Плата за склад — 40 Зені.";
			else mes "I'm sorry, but you don't have enough Zeny. Storage fee is 40 Zeny.";
			emotion e_cash;
			close;
		}
		set Zeny, Zeny - 40;
		set RESRVPTS, RESRVPTS + 4;
		if(#gl_lang == 1) mes "Зараз я відкрию ваше особисте сховище. Гарного дня~";
		else mes "Let me open your personal storage for you right away. Have a good day~";
		close2;
		openstorage;
		end;
	case 3:
		mes "[Cool Event Corp. Staff]";
		if(BaseClass != Job_Merchant){
			if(#gl_lang == 1) mes "Вибачте, оренда візка доступна лише для Торгових класів.";
			else mes "I'm sorry, but the Pushcart rental service is only available to Merchants.";
			emotion e_sry;
			close;
		}
		if(getskilllv(39) == 0){
			if(#gl_lang == 1) mes "Ви повинні спочатку вивчити навичку оренди візка.";
			else mes "You can only rent a cart after learning the Pushcart Skill.";
			close;
		}
		if(checkcart() == 1){
			if(#gl_lang == 1) mes "У вас вже є візок.";
			else mes "You already have a Pushcart equipped.";
			close;
		}
		if(countitem(7061)){
			delitem 7061,1;
			setcart;
			close;
		}
		if(#gl_lang == 1) mes "Плата за оренду візка — 800 Зені. Бажаєте орендувати?";
		else mes "The Pushcart rental fee is 800 Zeny. Would you like to rent a Pushcart?";
		next;
		if(select("Rent a Pushcart:Cancel") == 1){
			if(Zeny < 800){
				mes "[Cool Event Corp. Staff]";
				if(#gl_lang == 1) mes "У вас недостатньо грошей.";
				else mes "I'm sorry, but you don't have enough Zeny.";
				emotion e_cash;
				close;
			} else {
				set Zeny, Zeny - 800;
				set RESRVPTS, RESRVPTS + 80;
				setcart;
				close;
			}
		}
		close;
	case 4:
	default:
		mes "[Cool Event Corp. Staff]";
		if(#gl_lang == 1) mes "Дякуємо за ваші відгуки. Ми завжди прагнемо бути кращими.";
		else mes "Help us become the best by providing us with your opinions. Thank you.";
		close;
	}
}
