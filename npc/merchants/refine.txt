	M_Menu:

	menu getequipname(1),PART1,getequipname(2),PART2,getequipname(3),PART3,getequipname(4),PART4,getequipname(5),PART5,

		else { mes "Do you want me to refine your dumb brain?"; }
		emotion 6;
		goto M_Menu;
	PART2:
		set @part,2;
		if (getequipisequiped(2)) goto CHECK1;
		mes "[" + @name$ + "]";

		goto M_Menu;

		set @part,4;

		if (#gl_lang == 1) { mes "Я не можу перетворити твою праву руку на ультимативну зброю..."; }

	PART6:

	PART7:
		set @part,7;
		if (getequipisequiped(7)) goto CHECK1;

		if (#gl_lang == 1) { mes "Що ти маєш на увазі під аксесуаром? Який саме?"; }

		goto M_Menu;
	PART8:

		if (getequipisequiped(8)) goto CHECK1;
		mes "[" + @name$ + "]";
		if (#gl_lang == 1) { mes "Що ти маєш на увазі під аксесуаром? Який саме?"; }

		goto M_Menu;

	if(getequipisenableref(@part)) goto CHECK2;
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Я не можу працювати з цією річчю..."; }
	else { mes "I can't work on this item..."; }
	close;

CHECK2:
	if(getequipisidentify(@part)) goto CHECK3;
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Спочатку ви повинні ідентифікувати цей предмет."; }
	else { mes "You must appraise this item first."; }
	close;

CHECK3:
	if(getequiprefinerycnt(@part) < getbattleflag("refine_player_max")) goto REFINE0;	// [refine99] config-driven ceiling
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Ця зброя вже на максимальному рівні і більше не може бути покращена."; }
	else { mes "This weapon is already at its maximum level and can no longer be refined."; }
	close;

REFINE0:
	set @refineitemid, getequipid(@part);
	set @refinerycnt, getequiprefinerycnt(@part);
	if(getequipweaponlv(@part) > 0) goto REFINE1;
	set @matname$,"Elunium";
	set @material,985;
	set @price,2000;
	set @safe,4;
	if(@features == 1) goto L_refinefeatures;
	goto L_refinenormal;
REFINE1:
	if(getequipweaponlv(@part) > 1) goto REFINE2;
	set @matname$,"Phracon";
	set @material,1010;
	set @price,50;
	set @safe,7;
	if(@features == 1) goto L_refinefeatures;
	goto L_refinenormal;
REFINE2:
	if(getequipweaponlv(@part) > 2) goto REFINE3;
	set @matname$,"Emveretarcon";
	set @material,1011;
	set @price,200;
	set @safe,6;
	if(@features == 1) goto L_refinefeatures;
	goto L_refinenormal;

	if(getequipweaponlv(@part) > 3) goto REFINE4;
	set @matname$,"Oridecon";
	set @material,984;
	set @price,5000;
	set @safe,5;
	if(@features == 1) goto L_refinefeatures;
	goto L_refinenormal;
REFINE4:
	set @matname$,"Oridecon";
	set @material,984;
	set @price,20000;
	set @safe,4;
	if(@features == 1) goto L_refinefeatures;
	goto L_refinenormal;

L_refinenormal:

		mes "Для покращення цієї штуки мені потрібно ^ff9999" + @matname$ + "^000000 та оплата " + @price + " Зени.";
		mes "Продовжити?";
	} else {
		mes "To refine this stuff, I need ^ff9999" + @matname$ + "^000000 and the fee " + @price + " Zeny.";
		mes "Continue?";
	}
	next;
	if (#gl_lang == 1) { menu "Так",-,"Ні",Lcancel; }
	else { menu "Yes",-,"No",Lcancel; }
		if (getequippercentrefinery(@part) == 100) goto L_Sub;
		mes "[" + @name$ + "]";
		if (#gl_lang == 1) {
			mes "Хм... Зачекай! Це спорядження вже покращено до максимального рівня безпеки.";
			mes "Я маю попередити, якщо його покращувати ДАЛІ, воно може бути ЗНИЩЕНЕ та стати МАРНИМ!!";
		} else {
			mes "Hmm... Hold on! This piece of equipment has already been refined to its maximum safety level.";
			mes "I must warn you if it is refined ANYMORE, It could be DESTROYED and become USELESS!!";
		}
		next;
		mes "["+@name$+"]";
		if (#gl_lang == 1) { mes "Ви все ще хочете покращити його?"; }
		else { mes "Do you still wish you refine it?"; }
		next;
		if (#gl_lang == 1) { menu "Так",-,"Ні",Lcancel1; }
		else { menu "Yes",-,"No",Lcancel1; }
		L_Sub:
			if ((countitem(@material) < 1) || (Zeny < @price)) goto Lcancel2;
			set Zeny,Zeny-@price;
			delitem @material,1;
Lrefine:
	if (getequipisequiped(@part) == 0) goto LNoItem;
	if (getequipid(@part) != @refineitemid) goto LNoFake;
	if (getequiprefinerycnt(@part) != @refinerycnt) goto LNoFake;
	if (getequippercentrefinery(@part) <= rand(100)) goto Lfail;
	mes "["+@name$+"]";
	mes "Дзинь! Дзинь! Дзинь!";
	successrefitem @part;
	next;
	mes "["+@name$+"]";
	if (#gl_lang == 1) { mes "ХА-ХА! Схоже, мої навички ще не заіржавіли!"; }
	else { mes "HAHA! It seems my skills haven't gotten rusty yet!"; }
	emotion 21;
	close;

Lfail:
	mes "[" + @name$ + "]";
	mes "Дзинь! Дзинь! Дзинь!";
	failedrefitem @part;
	next;
	mes "["+@name$+"]";
	if (#gl_lang == 1) { mes "Аааах!! О ні...!!"; }
	else { mes "Aaahhh!! Oh no...!!"; }
	emotion 16;
	next;
	mes "["+@name$+"]";
	if (#gl_lang == 1) { mes "Мені дуже соромно... процес покращення ^ff0000провалився^000000."; }
	else { mes "I am deeply ashamed... refining process ^ff0000failed^000000."; }
	close;

	if (#gl_lang == 1) { mes "Дивись сюди... на тобі немає жодних речей..."; }
	else { mes "Look here... you don't have any Items on..."; }
	close;

LNoFake:
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Ти думав, що я такий дурний!?!"; }
	else { mes "Did you imagine I could be so stupid !?!"; }
	close;

Lcancel:
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Як скажеш... Хм, нехай буде так..."; }
	else { mes "You said so..Hmm so be it..."; }
	close;

Lcancel1:
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Гарний вибір."; }
	else { mes "Good Choice."; }
	close;

Lcancel2:
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Це все, що ти маєш? Я не можу працювати на тебе за меншу ціну."; }
	else { mes "Is that all you got? I can't work for you at a lower price."; }
	close;

L_refinefeatures:
	if(getequiprefinerycnt(@part) >= @safe) goto Lnosafe;
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Я можу покращити це до безпечної межі або певну кількість разів..."; }
	else { mes "I can refine this to the safe limit or a desired number of times..."; }

		menu "До безпечної межі, будь ласка.",Lsafe,"Я сам вирішу скільки разів.",Lnosafe,"Я передумав...",Lcancel;
	} else {
		menu "To the safe limit please.",Lsafe,"I'll decide how many times.",Lnosafe,"I've changed my mind...",Lcancel;
	}

Lsafe:
	set @refinecnt,@safe - getequiprefinerycnt(@part);
	set @fullprice,@price * @refinecnt;
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Це коштуватиме тобі " + @refinecnt + " " + @matname$ + " та " + @fullprice + " Зени. Згода?"; }
	else { mes "That will cost you " + @refinecnt + " " + @matname$ + " and " + @fullprice + " Zeny. Is that ok?"; }
	next;
	if (#gl_lang == 1) { menu "Так",-,"Ні...",Lcancel; }
	else { menu "Yes",-,"No...",Lcancel; }
	if((countitem(@material) < @refinecnt) || (Zeny < @fullprice)) goto Lcancel2;
	set Zeny,Zeny - @fullprice;
	delitem @material,@refinecnt;
	goto L_refinesafe;

Lnosafe:
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Отже, скільки разів ти хочеш, щоб я покращив твій предмет?"; }
	else { mes "So how many times would you like me to refine your item?"; }
	next;
	input @refinecnt;
	if (@refinecnt<1) goto Lcancel3;
	set @refinecheck,@refinecnt + getequiprefinerycnt(@part);
	if(@refinecheck > getbattleflag("refine_player_max")) goto Lcancel3;	// [refine99] config-driven ceiling
	set @fullprice,@price * @refinecnt;
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Це коштуватиме тобі " + @refinecnt + " " + @matname$ + " та " + @fullprice + " Зени... Згода?"; }
	else { mes "This will cost you " + @refinecnt + " " + @matname$ + " and " + @fullprice + " Zeny... Is that ok?"; }
	next;
	if (#gl_lang == 1) { menu "Так...",-,"Ні...",Lcancel; }
	else { menu "Yes...",-,"No...",Lcancel; }
		if(@refinecheck > @safe) goto Lwarn;
		if((countitem(@material) < @refinecnt) || (Zeny < @fullprice)) goto Lcancel2;
		set Zeny,Zeny - @fullprice;
		delitem @material,@refinecnt;
		goto L_refinenumber;
		end;

		Lwarn:
			set @refinecheck,@refinecheck - @safe;
			mes "[" + @name$ + "]";
			if (#gl_lang == 1) { mes "Це спроба покращити понад безпечну межу " + @refinecheck + " разів. Продовжити?"; }
			else { mes "This will try to refine past the safe limit " + @refinecheck + " times. Continue?"; }
			next;
			if (#gl_lang == 1) { menu "Так",-,"Ні...",Lcancel1; }
			else { menu "Yes",-,"No...",Lcancel1; }
				if((countitem(@material) < @refinecnt) || (Zeny < @fullprice)) goto Lcancel2;
				set Zeny,Zeny - @fullprice;
				delitem @material,@refinecnt;
				goto L_refinenumber;

Lcancel3:
	mes "[" + @name$ + "]";
	if (#gl_lang == 1) { mes "Я не можу покращити цей предмет таку кількість разів."; }
	else { mes "I can't refine this item that many times."; }
	close;

L_refinesafe:
	if (getequipisequiped(@part) == 0) goto LNoItem;
	if (getequipid(@part) != @refineitemid) goto LNoFake;
	if (getequippercentrefinery(@part) < 100) goto LNoFake;
	mes "Дзинь, дзинь!!!";
	successrefitem @part;
	emotion 21;
	set @refinecnt,@refinecnt - 1;
	next;
	if(@refinecnt == 0) goto Lend;
	goto L_refinesafe;

	Lend:
		mes "[" + @name$ + "]";
		if (#gl_lang == 1) { mes "Все готово... Приходь ще."; }
		else { mes "All finished... Come again soon."; }
		close;
		
L_refinenumber:
	if (getequipisequiped(@part) == 0) goto LNoItem;
	if (getequipid(@part) != @refineitemid) goto LNoFake;
	mes "Дзинь, дзинь!!!";
	if (getequippercentrefinery(@part)<=rand(100)) goto Lfail_number;
	successrefitem @part;
	emotion 21;
	set @refinecnt,@refinecnt - 1;
	next;
	if(@refinecnt == 0) goto Lend;
	goto L_refinenumber;

	Lfail_number:
		failedrefitem @part;
		emotion 23;
		mes "[" + @name$ + "]";
		if (#gl_lang == 1) { mes "ВАААА!!! Мені так шкода..."; }
		else { mes "WAHHHH!!! I'm so sorry..."; }
		set @refinecnt,@refinecnt - 1;
		if(@refinecnt == 0) goto Lend2;
		if (#gl_lang == 1) { mes "Ось невикористані Зени та матеріали назад..."; }
		else { mes "Here's the unused Zeny and Material back..."; }
		getitem @material,@refinecnt;
		set @fullprice,@refinecnt * @price;
		set Zeny,Zeny + @fullprice;
	Lend2:
		close;
}
