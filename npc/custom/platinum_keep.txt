//===== uaRO custom script ================================================
//= Platinum (quest) skill persistence
//===== Description: ======================================================
//= Records every platinum/quest skill a character learns and re-grants any
//= that are missing on login. The eAthena engine loses quest-skill grants
//= on a skill reset even with quest_skill_reset off (the platinum NPCs even
//= note it: "eA engine doesn't let you keep learn't skill"). This makes them
//= come back on the next login, so Find Stone / Stone Fling / Sonic
//= Acceleration / Throw Venom Knife etc. survive a reset. (S.)
//=
//= NB1: this engine has NO rAthena `parse_variable` -- it only understands
//= `set var,val;` assignment (binary operators like + & | << >> == DO parse).
//= The `=`-assignment and `++`/`+=` forms fail to parse, which silently drops
//= the whole function from userfunc_db ("callfunc: function not found!").
//= NB2: arrays here are ONLY allowed with a `$`, `@` or `.` prefix
//= (not_array_variable, script.c) -- there is NO permanent per-character
//= array. A bare `PLATINUM_SK[]` throws "getarraysize: illegal scope". So the
//= learned set is kept as a BITMASK over the fixed skill list, split across two
//= permanent per-character scalars PLATINUM_SK0 / PLATINUM_SK1 (array index =
//= bit; bits 0-31 -> SK0, 32+ -> SK1). The list is inlined in both blocks (no
//= passing an array into a function by reference). Keep classic syntax. (S.)
//=========================================================================

// Record that the character has learned platinum skill <id>: find its index in
// the fixed list and set that bit. Never reorder the list (index = bit); append
// only, and mirror any change to the OnPCLoginEvent copy below.
function	script	F_PlatRecord	{
	set .@id, getarg(0);
	setarray .@all[0],
		142,143,144,145,146,147,148,149,150,151,152,153,154,155,156,157,238,
		1001,1002,1003,1004,1005,1006,1007,1008,1009,1010,1011,1012,1013,1014,1015,1016,1017,1018,1019;
	set .@n, getarraysize(.@all);
	for (set .@i, 0; .@i < .@n; set .@i, .@i+1) {
		if (.@all[.@i] == .@id) {
			if (.@i < 32)
				set PLATINUM_SK0, PLATINUM_SK0 | (1 << .@i);
			else
				set PLATINUM_SK1, PLATINUM_SK1 | (1 << (.@i - 32));
			return;
		}
	}
	return;
}

// Grant a platinum skill AND remember it. The platinum NPCs call this instead
// of a bare `skill <id>,<lv>,0;` so the grant is recorded for re-grant on login.
function	script	F_KeepSkill	{
	set .@id, getarg(0);
	set .@lv, getarg(1);
	skill .@id,.@lv,0;
	callfunc "F_PlatRecord", .@id;
	return;
}

// On login: backfill the mask from skills the char already has, then re-grant
// any recorded platinum skill the character no longer has (e.g. wiped by a
// skill reset). All platinum skills are level 1.
-	script	PlatinumKeep	-1,{
OnPCLoginEvent:
	setarray .@all[0],
		142,143,144,145,146,147,148,149,150,151,152,153,154,155,156,157,238,
		1001,1002,1003,1004,1005,1006,1007,1008,1009,1010,1011,1012,1013,1014,1015,1016,1017,1018,1019;
	set .@n, getarraysize(.@all);
	for (set .@i, 0; .@i < .@n; set .@i, .@i+1) {
		if (.@i < 32)
			set .@bit, (PLATINUM_SK0 >> .@i) & 1;
		else
			set .@bit, (PLATINUM_SK1 >> (.@i - 32)) & 1;
		if (getskilllv(.@all[.@i]) >= 1) {
			// Backfill: has it but not recorded yet -> record for future resets.
			if (.@bit == 0) callfunc "F_PlatRecord", .@all[.@i];
		} else {
			// Recorded but lost (e.g. reset) -> restore it.
			if (.@bit == 1) skill .@all[.@i],1,0;
		}
	}
	end;
}
