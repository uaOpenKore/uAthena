//===== uaRO custom script ================================================
//= Platinum (quest) skill persistence
//===== Description: ======================================================
//= Records every platinum/quest skill a character learns and re-grants any
//= that are missing on login. The eAthena engine loses quest-skill grants
//= on a skill reset even with quest_skill_reset off (the platinum NPCs even
//= note it: "eA engine doesn't let you keep learn't skill"). This makes them
//= come back on the next login, so Find Stone / Stone Fling / Sonic
//= Acceleration / Throw Venom Knife etc. survive a reset. (S.)
//=
//= NB: this engine has NO rAthena `parse_variable` -- it only understands
//= `set var,val;` assignment and the binary `+` operator. The `=`-assignment
//= and `++` forms fail to parse, which silently drops the whole function from
//= userfunc_db ("callfunc: function not found! [F_KeepSkill]"). Keep this file
//= in classic eAthena syntax only: set / set .@i,.@i+1 / no ++ / no +=. (S.)
//=========================================================================

// Grant a platinum skill AND remember it. The platinum NPCs call this instead
// of a bare `skill <id>,<lv>,0;` so the grant is recorded in PLATINUM_SK[] (a
// saved, per-character array).
function	script	F_KeepSkill	{
	set .@id, getarg(0);
	set .@lv, getarg(1);
	skill .@id,.@lv,0;
	set .@n, getarraysize(PLATINUM_SK);
	for (set .@i, 0; .@i < .@n; set .@i, .@i+1)
		if (PLATINUM_SK[.@i] == .@id) return;   // already recorded -> done
	set PLATINUM_SK[.@n], .@id;
	return;
}

// On login, re-grant any recorded platinum skill the character no longer has
// (e.g. wiped by a skill reset). Level 1 -- all platinum skills are level 1.
-	script	PlatinumKeep	-1,{
OnPCLoginEvent:
	// Backfill: characters who learned a platinum skill BEFORE this system existed have it but it
	// isn't in PLATINUM_SK[] yet, so a future reset would lose it for good. On login, record every
	// platinum skill the char CURRENTLY has that isn't recorded yet. (This can't restore a skill
	// already lost before recording -- that needs one re-learn at the platinum NPC. S.)
	setarray .@all[0],
		142,143,144,145,146,147,148,149,150,151,152,153,154,155,156,157,238,
		1001,1002,1003,1004,1005,1006,1007,1008,1009,1010,1011,1012,1013,1014,1015,1016,1017,1018,1019;
	set .@na, getarraysize(.@all);
	for (set .@i, 0; .@i < .@na; set .@i, .@i+1) {
		if (getskilllv(.@all[.@i]) < 1) continue;       // don't have it -> nothing to record
		set .@have, 0;
		set .@nr, getarraysize(PLATINUM_SK);
		for (set .@j, 0; .@j < .@nr; set .@j, .@j+1)
			if (PLATINUM_SK[.@j] == .@all[.@i]) { set .@have, 1; break; }
		if (!.@have) set PLATINUM_SK[getarraysize(PLATINUM_SK)], .@all[.@i];  // record for future resets
	}
	// Re-grant: any recorded platinum skill the character no longer has (e.g. wiped by a reset).
	set .@n, getarraysize(PLATINUM_SK);
	for (set .@i, 0; .@i < .@n; set .@i, .@i+1)
		if (PLATINUM_SK[.@i] > 0 && getskilllv(PLATINUM_SK[.@i]) < 1)
			skill PLATINUM_SK[.@i],1,0;
	end;
}
