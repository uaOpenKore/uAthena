prontera,145,125,4	script	Mail	65,{
    mes "[Поштовий голуб]";
    mes "Привіт! Я можу передати твою річ іншому мандрівнику.";
    next;
    
    switch(select("Відправити предмет:Забрати пошту:Вихід")) {
    case 1:
        mes "[Поштовий голуб]";
        mes "Введіть точне ім'я отримувача:";
        input .@name$;
        
        // Перевірка, чи існує такий персонаж (залежить від версії емулятора)
        if (charid2rid(getcharid(0, .@name$)) == 0) {
            mes "Персонажа з таким ім'ям не знайдено в мережі.";
            // Примітка: Для офлайн-передачі потрібен запит до БД SQL
            close;
        }

        mes "Введіть ID предмета, який хочете відправити:";
        input .@itemid;
        
        if (countitem(.@itemid) < 1) {
            mes "У вас немає цього предмета!";
            close;
        }

        mes "Скільки штук ви хочете відправити?";
        input .@amount;

        if (countitem(.@itemid) < .@amount) {
            mes "У вас недостатньо предметів!";
            close;
        }

        // Логіка збереження (приклад через глобальні змінні, краще через SQL)
        // Тут спрощений варіант для демонстрації
        delitem .@itemid, .@amount;
        setgetvariable(strcharinfo(0), "mail_item_" + .@name$, .@itemid);
        setgetvariable(strcharinfo(0), "mail_count_" + .@name$, .@amount);
        
        mes "Предмет успішно відправлено!";
        close;

    case 2:
        // Перевірка наявності пошти для поточного гравця
        .@my_item = getvariable(strcharinfo(0), "mail_item_" + strcharinfo(0));
        .@my_count = getvariable(strcharinfo(0), "mail_count_" + strcharinfo(0));

        if (.@my_item == 0) {
            mes "Для вас немає нових повідомлень або речей.";
            close;
        }

        mes "Вам прийшла посилка: " + getitemname(.@my_item) + " (" + .@my_count + " шт.)";
        getitem .@my_item, .@my_count;
        
        // Очищення даних після отримання
        setgetvariable(strcharinfo(0), "mail_item_" + strcharinfo(0), 0);
        setgetvariable(strcharinfo(0), "mail_count_" + strcharinfo(0), 0);
        
        mes "Ви забрали свою посилку!";
        close;

    case 3:
        close;
    }
}
