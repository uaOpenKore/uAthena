
prontera,136,178,6	script	Card Exchanger	853,{

    set @cards_needed, 20;
    set @reward_id, 616; // Old Card Album

    if (#gl_lang == 1) {
    mes "[Обмінник карт]";
    mes "Привіт, я можу поміняти твої карти на карт альбом.";
    mes "З тебе ^FF0000"+@cards_needed+"^000000 любих карт,";
    mes "В замін я дам тобі новесенький ^0000FFOld Card Album^000000.";
}


    else if (#gl_lang == 0) {
    mes "[Card Exchanger]";
    mes "Hello. I am a card collector.";
    mes "If you give me ^FF0000"+@cards_needed+"^000000 of any cards,";
    mes "I will give you one ^0000FFOld Card Album^000000.";
}
    next;

    if (#gl_lang == 0) {
    menu "I have 20 cards to trade.",L_CheckCards, "No, thanks.",L_Close;
}

    else if (#gl_lang == 1) {
    menu "Я якраз маю 20 непотрібних карт.",L_CheckCards, "Ні, дякую.",L_Close;
}

L_CheckCards:

    set @inventory_cards, 0;
    set @i, 4001;

    while(@i < 4600) {
        if (countitem(@i) > 0) {
            set @inventory_cards, @inventory_cards + countitem(@i);
        }
        set @i, @i + 1;
        if (@inventory_cards >= @cards_needed) goto L_StartTrade;
    }

    if (#gl_lang == 0) {
    mes "[Card Exchanger]";
    mes "You don't have enough cards.";
    mes "I found only ^FF0000"+@inventory_cards+"^000000 cards.";
    mes "Please come back when you have at least "+@cards_needed+".";
}

    if (#gl_lang == 1) {
    mes "[Обмінник карт]";
    mes "В тебе невистачає карт.";
    mes "У тебе всього ^FF0000"+@inventory_cards+"^000000 шт.";
    mes "Приходь коли назбираєш "+@cards_needed+".";
}
    close;

L_StartTrade:
    if (#gl_lang == 0) {
    mes "[Card Exchanger]";
    mes "Excellent! I will take 20 of your cards now.";
}

    else if (#gl_lang == 1) {
    mes "[Обмінник карт]";
    mes "Чудово ось твій новенький Карт Альбом.";
}
    next;

    set @removed, 0;
    set @i, 4001;
    while(@removed < @cards_needed) {
        if (countitem(@i) > 0) {
            delitem @i, 1;
            set @removed, @removed + 1;
        } else {
            set @i, @i + 1;
        }
        if (@i > 5000) break; 
    }

    getitem @reward_id, 1;
    if (#gl_lang == 0) {
    mes "[Card Exchanger]";
    mes "Here is your album. Use it wisely!";
}

    else if (#gl_lang == 1) {
    mes "[Обмінник карт]";
    mes "Удачі тобі друже.";
}
    specialeffect2 154;
    close;

L_Close:
    if (#gl_lang == 0) {
    mes "[Card Exchanger]";
    mes "Fine. Many people want my albums anyway.";
}
    else if (#gl_lang == 1) {
    mes "[Обмінник карт]";
    mes "Ееее чудово. ";
    mes "Багато хто шукає мої альбоми.";
    mes "Йди поки цілий";
}

    close;
}
