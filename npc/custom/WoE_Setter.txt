//===== eAthena Script =======================================

//= Формат: IF == 1 (UA) / IF < 1 (EN)
//============================================================
prontera.gat,162,194,5	script	WoE Setter	806,{
	if (#gl_lang == 1) mes "[Налаштування WoE]";
	if (#gl_lang < 1) mes "[Woe Setter]";
	
	if(getgmlevel()<99) goto L_NotGM;
	
	if (#gl_lang == 1) {
		mes "Вітаю, я можу встановити час для WoE.";
		mes "Просто дотримуйтесь інструкцій у кожному розділі, і все буде готово до використання.";
	}
	if (#gl_lang < 1) {
		mes "Welcome, I can set WoE times for you";
		mes "Simply follow the intruction given in each section after this, and it should be set and ready to use";
	}
	next;
	
	if (#gl_lang == 1) {
		mes "[Налаштування WoE]";
		mes "Починаємо найцікавіше.";
		mes "Будь ласка, оберіть дію:";
	}
	if (#gl_lang < 1) {
		mes "[Woe Setter]";
		mes "Now the fun starts";
		mes "Please select what you want to do";
	}
	next;
	
	if (#gl_lang == 1) menu "Додати новий час WoE",L_Add,"Скинути ВСІ часи WoE",L_Reset,"Переглянути поточний розклад",-;
	if (#gl_lang < 1) menu "Add a WoE time",L_Add,"Reset All WoE times",L_Reset,"View Current WoE times",-;

	if (#gl_lang == 1) {
		mes "[Налаштування WoE]";
		mes "На даний момент встановлено такий розклад:";
	}
	if (#gl_lang < 1) {
		mes "[Woe Setter]";
		mes "Ok currently you have WoE times set for:-";
	}
	
	set @num,0;
L_LoopList:
	if($sday[@num] == 0 && $eday[@num] == 0 && $woetime[@num] == 0 && $woetime2[@num] == 0) goto L_LoopListEnd;
	
	if (#gl_lang == 1) {
		if($sday[@num]==0) set $@tempday$,"Неділя";
		if($sday[@num]==1) set $@tempday$,"Понеділок";
		if($sday[@num]==2) set $@tempday$,"Вівторок";
		if($sday[@num]==3) set $@tempday$,"Середа";
		if($sday[@num]==4) set $@tempday$,"Четвер";
		if($sday[@num]==5) set $@tempday$,"П'ятниця";
		if($sday[@num]==6) set $@tempday$,"Субота";
		
		if($eday[@num]==0) set $@tempday2$,"Неділя";
		if($eday[@num]==1) set $@tempday2$,"Понеділок";
		if($eday[@num]==2) set $@tempday2$,"Вівторок";
		if($eday[@num]==3) set $@tempday2$,"Середа";
		if($eday[@num]==4) set $@tempday2$,"Четвер";
		if($eday[@num]==5) set $@tempday2$,"П'ятниця";
		if($eday[@num]==6) set $@tempday2$,"Субота";
	}
	if (#gl_lang < 1) {
		if($sday[@num]==0) set $@tempday$,"Sunday";
		if($sday[@num]==1) set $@tempday$,"Monday";
		if($sday[@num]==2) set $@tempday$,"Tuesday";
		if($sday[@num]==3) set $@tempday$,"Wednesday";
		if($sday[@num]==4) set $@tempday$,"Thursday";
		if($sday[@num]==5) set $@tempday$,"Friday";
		if($sday[@num]==6) set $@tempday$,"Saturday";
		
		if($eday[@num]==0) set $@tempday2$,"Sunday";
		if($eday[@num]==1) set $@tempday2$,"Monday";
		if($eday[@num]==2) set $@tempday2$,"Tuesday";
		if($eday[@num]==3) set $@tempday2$,"Wednesday";
		if($eday[@num]==4) set $@tempday2$,"Thursday";
		if($eday[@num]==5) set $@tempday2$,"Friday";
		if($eday[@num]==6) set $@tempday2$,"Saturday";
	}

	if($eday[@num]==$sday[@num]) {
		if (#gl_lang == 1) mes $@tempday$+" з "+$woetime[@num]+":00 до "+$woetime2[@num]+":00";
		if (#gl_lang < 1) mes $@tempday$+" from "+$woetime[@num]+":00 till "+$woetime2[@num]+":00";
	}
	if($eday[@num]!=$sday[@num]) {
		if (#gl_lang == 1) mes "З "+$@tempday$+" "+$woetime[@num]+":00 до "+$@tempday2$+" "+$woetime2[@num]+":00";
		if (#gl_lang < 1) mes "From "+$@tempday$+" "+$woetime[@num]+":00 till "+$@tempday2$+" "+$woetime2[@num]+":00";
	}
	
	set @num,@num+1;
	goto L_LoopList;

L_LoopListEnd:
	close;

L_Add:
	set @num,0;
L_AddLoop:
	if($sday[@num] == 0 && $eday[@num] == 0 && $woetime[@num] == 0 && $woetime2[@num] == 0) goto L_AddLoopEnd;
	set @num,@num+1;
	goto L_AddLoop;

L_AddLoopEnd: 
	if (#gl_lang == 1) {
		mes "[Налаштування WoE]";
		mes "У який день WoE повинна ^FF0000початися^000000?";
		next;
		menu "Неділя",-,"Понеділок",-,"Вівторок",-,"Середа",-,"Четвер",-,"П'ятниця",-,"Субота",-;
		set $sday[@num],@menu-1;
		mes "[Налаштування WoE]";
		mes "Добре, тепер введіть годину ^FF0000початку^000000.";
		mes "Використовуйте 24-годинний формат (00-23):";
	}
	if (#gl_lang < 1) {
		mes "[Woe Setter]";
		mes "What day do you want the WoE to start ^FF0000start^000000 in?";
		next;
		menu "Sunday",-,"Monday",-,"Tuesday",-,"Wednesday",-,"Thursday",-,"Friday",-,"Saturday",-;
		set $sday[@num],@menu-1;
		mes "[Woe Setter]";
		mes "Ok, now please enter the hour you wish the WoE to ^FF0000start^000000 at";
		mes "This is using a 24 hour clock";
	}
	next;
	input @input;
	if (@input < 0 || @input > 23) goto L_BadNumber;
	set $woetime[@num],@input;

	if (#gl_lang == 1) {
		mes "[Налаштування WoE]";
		mes "У який день WoE повинна ^FF0000закінчитися^000000?";
		next;
		menu "У той самий день",-,"Неділя",-,"Понеділок",-,"Вівторок",-,"Середа",-,"Четвер",-,"П'ятниця",-,"Субота",-;
	}
	if (#gl_lang < 1) {
		mes "[Woe Setter]";
		mes "What day do you want the WoE to start ^FF0000finish^000000 in?";
		next;
		menu "Same as start",-,"Sunday",-,"Monday",-,"Tuesday",-,"Wednesday",-,"Thursday",-,"Friday",-,"Saturday",-;
	}
	if(@menu==1) set $eday[@num],$sday[@num];
	if(@menu>1) set $eday[@num],@menu-2;

	if (#gl_lang == 1) {
		mes "[Налаштування WoE]";
		mes "Добре, тепер введіть годину ^FF0000завершення^000000.";
	}
	if (#gl_lang < 1) {
		mes "[Woe Setter]";
		mes "Ok, now please enter the hour you wish the WoE to ^FF0000finish^000000 at";
	}
	next;
	input @input;
	if (@input < 0 || @input > 23) goto L_BadNumber;
	set $woetime2[@num],@input;

	if (#gl_lang == 1) {
		mes "[Налаштування WoE]";
		mes "Час WoE встановлено!";
		mes "Щоб підтвердити, оберіть пункт ^FF0000'Переглянути поточний розклад'^000000.";
	}
	if (#gl_lang < 1) {
		mes "[Woe Setter]";
		mes "You WoE time has now been set";
		mes "To confirm this, ask me to show you the ^FF0000'View Current WoE times'^000000";
	}
	close;

L_Reset:
	if (#gl_lang == 1) {
		mes "[Налаштування WoE]";
		mes "Ви збираєтеся видалити весь створений розклад WoE.";
		mes "Ви впевнені?";
		next;
		menu "Ні, я передумав",L_No,"Так, я дійсно хочу це зробити",-;
	}
	if (#gl_lang < 1) {
		mes "[Woe Setter]";
		mes "You are about to reset all the set WoE times you have created";
		mes "Are you sure?";
		next;
		menu "No, what was I thinking",L_No,"Yes I really want to do it",-;
	}
	
	if(agitcheck()) goto L_WoEOn;
	deletearray $sday[0],200;
	deletearray $eday[0],200;
	deletearray $woetime[0],200;
	deletearray $woetime2[0],200;	
	
	if (#gl_lang == 1) mes "Тепер розклад порожній. Не забудьте встановити новий.";
	if (#gl_lang < 1) mes "They are all gone now, please remember to set new ones";
	close;

L_WoEOn:
	if (#gl_lang == 1) mes "Вибачте, поки триває WoE, ви не можете скинути розклад.";
	if (#gl_lang < 1) mes "Sorry since there is a WoE in progress you cannot reset the WoE times";
	close;

L_No:
	if (#gl_lang == 1) mes "О, добре. Заходьте пізніше.";
	if (#gl_lang < 1) mes "Oh, good, come back whenever";
	close;

L_NotGM:
	if (#gl_lang == 1) mes "Поточний розклад WoE:-";
	if (#gl_lang < 1) mes "The current WoE times are :-";
	set @num,0;
	goto L_LoopList;

L_BadNumber:
	if (#gl_lang == 1) mes "Вибачте, ви ввели неправильне число. Спробуйте ще раз.";
	if (#gl_lang < 1) mes "Sorry that was an invalid number, please try again";
	close;
}

-	script	Agit_Event2	-1,{
	end;

OnInit:
	disablenpc "Agit_Event";
OnMinute00:
OnAgitInit:
	set $@num,0;
L_StartLoop:
	if($sday[$@num] == 0 && $eday[$@num] == 0 && $woetime[$@num] == 0 && $woetime2[$@num] == 0) goto L_StartLoopEnd;
	if(gettime(4)==$sday[$@num] && gettime(3)>=$woetime[$@num] && (gettime(3)<$woetime2[$@num] || $sday[$@num]!=$eday[$@num])) goto L_Start;
	set $@num,$@num+1;
	goto L_StartLoop;

L_StartLoopEnd:
	set $@num,0;
L_EndLoop:
	if($eday[$@num] == 0 && $woetime[$@num] == 0 && $woetime2[$@num] == 0) end;
	if((gettime(4)==$eday[$@num]) && (gettime(3)==$woetime2[$@num])) goto L_End;
	set $@num,$@num+1;
	goto L_EndLoop;
	end;

L_End:
	if(!agitcheck()) end;
	if (#gl_lang == 1) announce "Війна за Імперіум закінчилася!",8;
	if (#gl_lang < 1) announce "The War Of Emperium is over!",8;
	agitend;
	end;

L_Start:
	if(agitcheck()) end;
	if (#gl_lang == 1) announce "Війна за Імперіум розпочалася!",8;
	if (#gl_lang < 1) announce "The War Of Emperium has begun!",8;
	agitstart;
	end;
}
