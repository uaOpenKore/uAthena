umbala,67,144,5	script	Master Rodrigo	842,{
	if (capoeira_quest == 4) {
		mes "[Master Rodrigo]";
		if (#gl_lang == 1) mes "Твій танець вільний. Ти справжній майстер.";
		else mes "Your dance is free. You are a true master.";
		close;
	}

	if (killed_wanderer == 1) {
		mes "[Master Rodrigo]";
		if (#gl_lang == 1) {
			mes "Ти подолав свій страх. Тепер ти вільний.";
			mes "Прийми цю зброю як символ твого звільнення.";
		} else {
			mes "You have overcome your fear. Now you are free.";
			mes "Accept this weapon as a symbol of your liberation.";
		}
		getitem 1813,1; // Kaiser Knuckle
		set capoeira_quest, 4;
		set killed_wanderer, 0;
		close;
	}

	if (capoeira_quest == 2) goto L_CheckKill;
	if (capoeira_quest == 1) goto L_WaitTraining;
	
	mes "[Master Rodrigo]";
	if (#gl_lang == 1) mes "Чи чув ти про ^FF0000Капоейру^000000? Це танець свободи.";
	else mes "Have you heard of ^FF0000Capoeira^000000? It is the dance of freedom.";
	next;
	
	if (#gl_lang == 1) {
		menu "Розкажи мені про це.",L_StartQuest,"Мені це не цікаво.",L_Exit;
	} else {
		menu "Tell me about it.",L_StartQuest,"I'm not interested.",L_Exit;
	}

L_StartQuest:
	mes "[Master Rodrigo]";
	if (#gl_lang == 1) mes "Знайди трьох старійшин в Умбалі. Вони розкажуть про наші коріння.";
	else mes "Find three elders in Umbala. They will tell you about our roots.";
	set capoeira_quest, 1;
	set capoeira_story, 0; 
	set q_amaru, 0; set q_kaua, 0; set q_tiago, 0;
	close;

L_WaitTraining:
	if (capoeira_story < 3) {
		mes "[Master Rodrigo]";
		if (#gl_lang == 1) mes "Ти ще не розмовляв з усіма старійшинами ("+capoeira_story+"/3).";
		else mes "You haven't spoken to all the elders yet ("+capoeira_story+"/3).";
		close;
	}
	mes "[Master Rodrigo]";
	if (#gl_lang == 1) mes "Тепер знайди трьох майстрів у ^0000FFКомодо^000000, вони навчать тебе рухам.";
	else mes "Now find three masters in ^0000FFComodo^000000, they will teach you the moves.";
	set capoeira_quest, 2;
	set capoeira_train, 0;
	set q_m1, 0; set q_m2, 0; set q_m3, 0;
	close;

L_CheckKill:
	if (capoeira_train < 3) {
		mes "[Master Rodrigo]";
		if (#gl_lang == 1) mes "Майстри в Комодо ще не закінчили навчання ("+capoeira_train+"/3).";
		else mes "The masters in Comodo haven't finished training you ("+capoeira_train+"/3).";
		close;
	}
	mes "[Master Rodrigo]";
	if (#gl_lang == 1) mes "Іди на берег Комодо (79, 344) і подолай духа загарбника — ^FF0000Wander Man^000000!";
	else mes "Go to the shores of Comodo (79, 344) and defeat the invader spirit — ^FF0000Wander Man^000000!";
	close;

L_Exit:
	close;
}

// --- SPAWN TRIGGER (Wander Man) ---
comodo,79,344,0	script	CapoeiraTrig	-1,9,9,{
OnTouch:
	if (capoeira_quest == 2 && capoeira_train == 3 && killed_wanderer != 1) {
		set killed_wanderer, 2; 
		killmonster "comodo","CapoeiraTrig::OnBossKill"; 
		monster "comodo",79,344,"Wander Man",1208,1,"CapoeiraTrig::OnBossKill";
		
		if (#gl_lang == 1) announce "Wander Man: Ти повернувся? Твоя душа буде моєю!",bc_self;
		else announce "Wander Man: You returned? Your soul will be mine!",bc_self;
	}
	end;

OnBossKill:
	set killed_wanderer, 1;
	if (#gl_lang == 1) dispbottom "Wander Man подоланий! Поверніться до Родріго за нагородою.";
	else dispbottom "Wander Man defeated! Return to Rodrigo for your reward.";
	end;
}
