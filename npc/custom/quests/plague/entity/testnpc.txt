prontera,151,142,5	script	VarManager	65,{

	mes "[Менеджер]";
	mes "Яку змінну перевірити?";
	next;

	// В uAthena 10.3 краще спочатку обнулити тимчасову строку
	set .@n$,"";
	input .@n$;

	if(.@n$ == "") {
		mes "Помилка назви.";
		close;
	}

	// Отримуємо значення
	set .@v, getd(.@n$);

	mes "Змінна: " + .@n$;
	mes "Значення: ^FF0000" + .@v + "^000000";
	next;

	menu "Змінити",L_Edit,"Вихід",L_Quit;

L_Edit:
	mes "Введіть нове число:";
	next;
	input .@v;
	
	setd .@n$, .@v;
	
	mes "Готово. Нове значення: " + getd(.@n$);
	close;

L_Quit:
	close;
}
