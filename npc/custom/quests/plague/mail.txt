prontera,164,171,4	script	Mail Service	888,{

    set .@L, #gl_lang; // 1 - UA, 0 - EN

    mes "[Mail Service]";
    if (.@L == 1) mes "Привіт, ^0000FF" + strcharinfo(0) + "^000000!";
    else mes "Hello, ^0000FF" + strcharinfo(0) + "^000000!";
    next;
    
    if (.@L == 1) set .@menu$, "Перевірити скриньку:Надіслати пошту:Вихід";
    else set .@menu$, "Check Inbox:Send Mail:Exit";
    
    set @menu, select(.@menu$);
    if (@menu == 3) close;

    // --- ОТРИМАННЯ (INBOX) ---
    if (@menu == 1) {
        deletearray @m_id, 10; deletearray @s_name$, 10; deletearray @m_z, 10;
        deletearray @i1, 10; deletearray @a1, 10; deletearray @i2, 10; deletearray @a2, 10; deletearray @i3, 10; deletearray @a3, 10;

        set @my_aid, getcharid(3);
        query_sql "SELECT id, sender_name, zeny, id1, am1, id2, am2, id3, am3 FROM `custom_mail` WHERE dest_id="+@my_aid+" LIMIT 10", @m_id, @s_name$, @m_z, @i1, @a1, @i2, @a2, @i3, @a3;
        
        if (getarraysize(@m_id) == 0) { 
            if (.@L == 1) mes "Ваша скринька порожня.";
            else mes "Your inbox is empty.";
            close; 
        }

        set @m_str$, "";
        for(set @i, 0; @i < getarraysize(@m_id); set @i, @i + 1) {
            if (.@L == 1) set @m_str$, @m_str$ + "Отримати від " + @s_name$[@i] + ":";
            else set @m_str$, @m_str$ + "Receive from " + @s_name$[@i] + ":";
        }
        
        set @s, select(@m_str$) - 1;
        
        if (.@L == 1) mes "[Лист від " + @s_name$[@s] + "]";
        else mes "[Mail from " + @s_name$[@s] + "]";
        
        if (@m_z[@s] > 0) mes " > Zeny: " + @m_z[@s];
        if (@i1[@s] > 0) mes " > " + getitemname(@i1[@s]) + " (" + @a1[@s] + ")";
        if (@i2[@s] > 0) mes " > " + getitemname(@i2[@s]) + " (" + @a2[@s] + ")";
        if (@i3[@s] > 0) mes " > " + getitemname(@i3[@s]) + " (" + @a3[@s] + ")";
        next;

        if (.@L == 1) set .@inbox_op$, "Забрати все:Назад";
        else set .@inbox_op$, "Claim all:Back";
        
        if (select(.@inbox_op$) == 2) close;

        set Zeny, Zeny + @m_z[@s];
        if (@i1[@s] > 0) getitem @i1[@s], @a1[@s];
        if (@i2[@s] > 0) getitem @i2[@s], @a2[@s];
        if (@i3[@s] > 0) getitem @i3[@s], @a3[@s];
        
        query_sql "DELETE FROM `custom_mail` WHERE id="+@m_id[@s];
        
        if (.@L == 1) mes "Вміст отримано. Лист видалено.";
        else mes "Items claimed. Mail deleted.";
        close;
    }

    // --- ВІДПРАВКА (SEND) ---
    if (@menu == 2) {
        if (.@L == 1) mes "Кому надіслати?";
        else mes "Who is the recipient?";
        input @target$;
        query_sql "SELECT account_id FROM `char` WHERE name='"+@target$+"' LIMIT 1", @t_aid;
        
        if (@t_aid == 0) { 
            if (.@L == 1) mes "Гравця ^FF0000" + @target$ + "^000000 не знайдено.";
            else mes "Player ^FF0000" + @target$ + "^000000 not found.";
            close; 
        }

        if (.@L == 1) mes "Скільки Zeny?";
        else mes "How much Zeny?";
        input @z;
        if (@z > Zeny || @z < 0) { 
            if (.@L == 1) mes "Недостатньо грошей."; 
            else mes "Not enough Zeny.";
            close; 
        }

        set @sid1,0; set @sam1,0; set @sid2,0; set @sam2,0; set @sid3,0; set @sam3,0;
        set @c, 1;

        while(@c <= 3) {
            if (.@L == 1) mes "Додати предмет №" + @c + "?";
            else mes "Add item #" + @c + "?";
            
            if (.@L == 1) set .@add$, "Так, обрати:Ні, досить";
            else set .@add$, "Yes, pick item:No, that's it";
            
            if (select(.@add$) == 2) break;

            getinventorylist;
            set @ms$, ""; set @vc, 0;
            deletearray @vids, 100;

            for(set @j, 0; @j < @inventorylist_count; set @j, @j + 1) {
                if (@inventorylist_equip[@j] == 0 && (getiteminfo(@inventorylist_id[@j], 6) & 1) == 0) {
                    set @ms$, @ms$ + getitemname(@inventorylist_id[@j]) + " [" + @inventorylist_amount[@j] + "]:";
                    set @vids[@vc], @inventorylist_id[@j];
                    set @vc, @vc + 1;
                }
            }
            if (@vc == 0) { 
                if (.@L == 1) mes "Немає доступних речей."; 
                else mes "No available items.";
                break; 
            }
            if (.@L == 1) set @ms$, @ms$ + "^FF0000Скасувати^000000";
            else set @ms$, @ms$ + "^FF0000Cancel^000000";
            
            set @sel, select(@ms$) - 1;
            if (@sel == @vc) continue;

            set @cid, @vids[@sel];
            if (.@L == 1) mes "Скільки " + getitemname(@cid) + "? (Макс: " + countitem(@cid) + ")";
            else mes "How many " + getitemname(@cid) + "? (Max: " + countitem(@cid) + ")";
            input @ca;
            if (@ca <= 0 || @ca > countitem(@cid)) { 
                if (.@L == 1) mes "Помилка кількості."; 
                else mes "Invalid amount.";
                close; 
            }

            if (@c == 1) { set @sid1, @cid; set @sam1, @ca; }
            if (@c == 2) { set @sid2, @cid; set @sam2, @ca; }
            if (@c == 3) { set @sid3, @cid; set @sam3, @ca; }

            delitem @cid, @ca;
            set @c, @c + 1;
        }

        if (.@L == 1) mes "Відправити лист для ^0000FF" + @target$ + "^000000?";
        else mes "Send mail to ^0000FF" + @target$ + "^000000?";
        
        if (.@L == 1) set .@send_op$, "Так, відправити:Скасувати все";
        else set .@send_op$, "Yes, send:Cancel all";
        
        if (select(.@send_op$) == 2) {
            if (@sid1>0) getitem @sid1, @sam1;
            if (@sid2>0) getitem @sid2, @sam2;
            if (@sid3>0) getitem @sid3, @sam3;
            close;
        }

        set Zeny, Zeny - @z;
        query_sql "INSERT INTO `custom_mail` (sender_name, dest_id, zeny, id1, am1, id2, am2, id3, am3) VALUES ('"+strcharinfo(0)+"', "+@t_aid+", "+@z+", "+@sid1+", "+@sam1+", "+@sid2+", "+@sam2+", "+@sid3+", "+@sam3+")";
        
        if (.@L == 1) mes "Посилка відправлена!";
        else mes "Mail sent successfully!";
        close;
    }
}

OnPCLoginEvent:
	query_sql "SELECT COUNT(*) FROM `custom_mail` WHERE dest_id="+getcharid(3), @mail_count;
	if (@mail_count > 0) {
		sleep2 3000;
        set .@L, #gl_lang;
		soundeffect "quest_alert.wav", 0; 
        if (.@L == 1) dispbottom "Пошта: У вас є " + @mail_count + " нових листів!";
		else dispbottom "Mail: You have " + @mail_count + " new messages!";
		emotion 21; 
	}
	end;
}
