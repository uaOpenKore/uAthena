// Card removal NPC by TyrNemesis^
// DANGEROUS! TODO: Think.. think 8) [Lupus]

prt_in.gat,28,73,4	script	Wise Old Woman	78,{

UPGRADEROOT:
	if (#gl_lang == 1) {
		mes "[Мудра бабуся]";
		mes "Доброго дня, юначе. У моїх силах вилучити карти, які ти вставив у своє спорядження. Тобі цікава така пропозиція?";
	}
	if (#gl_lang < 1) {
		mes "[Wise Old Woman]";
		mes "Good day, young one. I have the power to remove cards that you have compounded onto your equipment. Does this idea please you?";
	}
	next;
	
	if (#gl_lang == 1) menu "Так, цікава.",REMOVEMENU,"Скільки це коштує?",REMOVEPRICE,"Ні, дякую.",CLOSEOUT;
	if (#gl_lang < 1) menu "Yes, it does.",REMOVEMENU,"What do you charge?",REMOVEPRICE,"No thanks.",CLOSEOUT;

REMOVEPRICE:
	if (#gl_lang == 1) {
		mes "[Мудра бабуся]";
		mes "Я беру фіксовану плату у 200,000 зені, плюс 25,000 зені за кожну вилучену карту. Крім того, мені потрібні Star Crumb та Yellow Gemstone, щоб моє закляття спрацювало.";
	}
	if (#gl_lang < 1) {
		mes "[Wise Old Woman]";
		mes "I charge a flat fee of 200000 zeny, plus 25000 zeny for each card I remove from the item. In addition, I need a star crumb and a yellow gemstone to work my magic.";
	}
	next;
	
	if (#gl_lang == 1) menu "Добре. Зробімо це.",REMOVEMENU,"Ні, дякую.",CLOSEOUT;
	if (#gl_lang < 1) menu "Very well. Let's do it.",REMOVEMENU,"No thanks.",CLOSEOUT;

REMOVEMENU:
	if (#gl_lang == 1) {
		mes "[Мудра бабуся]";
		mes "Дуже добре. Яку річ мені оглянути?";
	}
	if (#gl_lang < 1) {
		mes "[Wise Old Woman]";
		mes "Very well. Which item shall I examine for you?";
	}
	next;
	
	if (#gl_lang == 1) set .@cancel, "Я передумав.";
	if (#gl_lang < 1) set .@cancel, "I changed my mind.";

	menu .@cancel,CLOSEOUT,
	     getequipname(1),SLOT1,
	     getequipname(2),SLOT2,
	     getequipname(3),SLOT3,
	     getequipname(4),SLOT4,
	     getequipname(5),SLOT5,
	     getequipname(6),SLOT6,
	     getequipname(7),SLOT7,
	     getequipname(8),SLOT8,
	     getequipname(9),SLOT9,
	     getequipname(10),SLOT10;

SLOT1: set @part,1; goto CARDNUMCHECK;
SLOT2: set @part,2; goto CARDNUMCHECK;
SLOT3: set @part,3; goto CARDNUMCHECK;
SLOT4: set @part,4; goto CARDNUMCHECK;
SLOT5: set @part,5; goto CARDNUMCHECK;
SLOT6: set @part,6; goto CARDNUMCHECK;
SLOT7: set @part,7; goto CARDNUMCHECK;
SLOT8: set @part,8; goto CARDNUMCHECK;
SLOT9: set @part,9; goto CARDNUMCHECK;
SLOT10: set @part,10; goto CARDNUMCHECK;

CARDNUMCHECK:
	if(getequipcardcnt(@part) == 0) goto DENYCARDCOUNT;
	set @cardcount,getequipcardcnt(@part);
	if(@cardcount > 1) goto CARDNUMMULTIMSG;
	
	if (#gl_lang == 1) {
		mes "[Мудра бабуся]";
		mes "Ця річ має " + @cardcount + " карту. Для ритуалу мені знадобиться 225,000 зені, ^0000FFStar Crumb^000000 та ^0000FFYellow Gemstone^000000.";
	}
	if (#gl_lang < 1) {
		mes "[Wise Old Woman]";
		mes "This item has " + @cardcount + " card compounded on it. To perform my magic, I will need 225000 zeny, a ^0000FFStar Crumb^000000, and a ^0000FFYellow Gemstone^000000.";
	}
	goto CARDNUMPOSTMSG;

CARDNUMMULTIMSG:
	if (#gl_lang == 1) {
		mes "[Мудра бабуся]";
		mes "Ця річ має " + @cardcount + " карти. Для ритуалу мені знадобиться " + (200000+(@cardcount * 25000)) + " зені, ^0000FFStar Crumb^000000 та ^0000FFYellow Gemstone^000000.";
	}
	if (#gl_lang < 1) {
		mes "[Wise Old Woman]";
		mes "This item has " + @cardcount + " cards compounded on it. To perform my magic, I will need " + (200000+(@cardcount * 25000)) + " zeny, a ^0000FFStar Crumb^000000, and a ^0000FFYellow Gemstone^000000.";
	}

CARDNUMPOSTMSG:
	next;
	if (#gl_lang == 1) menu "Добре. Роби.",REMOVECARDWARNING,"Я передумав.",CLOSEOUT;
	if (#gl_lang < 1) menu "Very well. Do it.",REMOVECARDWARNING,"Never mind.",CLOSEOUT;

REMOVECARDWARNING:
	if (#gl_lang == 1) {
		mes "[Мудра бабуся]";
		mes "Перш ніж я почну, мушу попередити — я можу зазнати невдачі. Якщо це станеться, я можу знищити карти, річ або і те, і інше. Гроші я не повертаю. Скажи мені, що для тебе важливіше: Карти чи Річ?";
	}
	if (#gl_lang < 1) {
		mes "[Wise Old Woman]";
		mes "Before I begin, I must warn you--I may fail. If I do, I may destroy the cards, the item, or both. I do not give refunds. That being said, which is more important to you: The cards, or the item?";
	}
	next;
	if (#gl_lang == 1) menu "Я передумав.",CLOSEOUT,"Річ.",PRIORITYITEM,"Карти.",PRIORITYCARD;
	if (#gl_lang < 1) menu "I changed my mind about this.",CLOSEOUT,"The item.",PRIORITYITEM,"The cards.",PRIORITYCARD;

PRIORITYITEM: set @failtype,1; goto REMOVECARD;
PRIORITYCARD: set @failtype,2; goto REMOVECARD;

REMOVECARD:
	if (#gl_lang == 1) mes "[Мудра бабуся]";
	if (#gl_lang < 1) mes "[Wise Old Woman]";
	if (#gl_lang == 1) mes "Дуже добре. Я починаю.";
	if (#gl_lang < 1) mes "Very well. I shall begin.";

	if((zeny < (200000+(@cardcount * 25000))) || (countitem(1000) < 1) || (countitem(715) < 1)) goto DENYMATERIAL;
	set zeny,zeny - (200000+(@cardcount * 25000));
	delitem 1000,1;
	delitem 715,1;

	set @failchance,rand(100);
	if(@failchance < 10) goto FAILREMOVECARD3;
	successremovecards @part;
	next;
	if (#gl_lang == 1) mes "[Мудра бабуся]";
	if (#gl_lang < 1) mes "[Wise Old Woman]";
	if (#gl_lang == 1) mes "Процес пройшов успішно. Ось твої карти та твоя річ. Прощавай.";
	if (#gl_lang < 1) mes "The process was a success. Here are your cards and your item. Farewell.";
	close;

FAILREMOVECARD0:
	failedremovecards @part,0;
	next;
	if (#gl_lang == 1) mes "[Мудра бабуся]";
	if (#gl_lang < 1) mes "[Wise Old Woman]";
	if (#gl_lang == 1) mes "Це повний провал. Боюся, що і річ, і карти були знищені.";
	if (#gl_lang < 1) mes "The process was a total failure. I am afraid the item and the cards were destroyed.";
	close;

FAILREMOVECARD1:
	failedremovecards @part,1;
	next;
	if (#gl_lang == 1) mes "[Мудра бабуся]";
	if (#gl_lang < 1) mes "[Wise Old Woman]";
	if (#gl_lang == 1) mes "Мені вдалося вилучити карти з речі, але вони були знищені в процесі. Проте з річчю все гаразд.";
	if (#gl_lang < 1) mes "While I have managed to remove the cards from the item, they were destroyed in the process. The item, however, is okay.";
	close;

FAILREMOVECARD2:
	failedremovecards @part,2;
	next;
	if (#gl_lang == 1) mes "[Мудра бабуся]";
	if (#gl_lang < 1) mes "[Wise Old Woman]";
	if (#gl_lang == 1) mes "Яка прикрість. Я змогла вилучити карти, але сама річ була знищена в процесі.";
	if (#gl_lang < 1) mes "Most unfortunate. I succeeded at removing the cards, but the item itself was destroyed in the process.";
	close;

FAILREMOVECARD3:
	failedremovecards @part,3;
	next;
	if (#gl_lang == 1) mes "[Мудра бабуся]";
	if (#gl_lang < 1) mes "[Wise Old Woman]";
	if (#gl_lang == 1) mes "Мені не вдалося вилучити карти. На щастя, і річ, і карти залишилися неушкодженими.";
	if (#gl_lang < 1) mes "I have failed to remove the cards. Luckily, however, both the item and the cards are still okay.";
	close;

DENYCARDCOUNT:
	if (#gl_lang == 1) {
		mes "[Мудра бабуся]";
		mes "Юначе... У цій речі немає вставлених карт. Боюся, я нічим не можу допомогти.";
	}
	if (#gl_lang < 1) {
		mes "[Wise Old Woman]";
		mes "Young one... There are no cards compounded on this item. I can do nothing with it, I'm afraid.";
	}
	close;

DENYMATERIAL:
	next;
	if (#gl_lang == 1) {
		mes "[Мудра бабуся]";
		mes "У тебе немає всіх необхідних предметів для мого закляття, дитино. Приходь, коли знайдеш їх.";
	}
	if (#gl_lang < 1) {
		mes "[Wise Old Woman]";
		mes "You do not have all the items I require to work my magic, child. Come again when you do.";
	}
	close;

CLOSEOUT:
	if (#gl_lang == 1) {
		mes "[Мудра бабуся]";
		mes "Дуже добре. Повертайся негайно, якщо тобі знову знадобляться мої послуги.";
	}
	if (#gl_lang < 1) {
		mes "[Wise Old Woman]";
		mes "Very well. Return at once if you seek my services.";
	}
	close;
}
