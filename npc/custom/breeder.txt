

prontera.gat,122,200,1	script	Universal Rental Npc	726,{
	if (#gl_lang == 1) {
		mes "[Універсальна оренда]";
		mes "Привіт, тут ти можеш орендувати візки, соколів або пекопеко.";
	}
	if (#gl_lang < 1) {
		mes "[Universal Rental Npc]";
		mes "Hi, here you can rent Carts, Falcons or Pecopecos.";
	}
	next;

L_Menu:
	if (#gl_lang == 1) menu "Орендувати візок",L_Cart,"Орендувати сокола",L_Falcon,"Орендувати Пекопеко",L_Peco,"Вийти",L_Quit;
	if (#gl_lang < 1) menu "Rent a Cart",L_Cart,"Rent a Falcon",L_Falcon,"Rent a Pecopeco",L_Peco,"Quit",L_Quit;
	close;

L_Cart:
	if((BaseClass == Job_Merchant || BaseJob == Job_SuperNovice) && checkcart() == 0) goto L_Cart_Ok;
	if(getskilllv(39)<1) goto L_Need_Skill;

	if (#gl_lang == 1) {
		mes "[Універсальна оренда]";
		mes "Вибач, " + strcharinfo(0) + ", але я здаю візки лише представникам професії Торговець, які мають достатньо навичок, щоб ними керувати.";
	}
	if (#gl_lang < 1) {
		mes "[Universal Rental Npc]";
		mes "Sorry " + strcharinfo(0) + ", but I only rent carts to people with the Merchant job root, who have enough skills to handle a cart.";
	}
	close;

L_Cart_Ok:
	setcart;
	goto L_Quit2;

L_Need_Skill:
	if (#gl_lang == 1) {
		mes "[Універсальна оренда]";
		mes "Вибач, у тебе немає необхідної навички для оренди візка.";
	}
	if (#gl_lang < 1) {
		mes "[Universal Rental Npc]";
		mes "Sorry you don't have the required skill to rent a cart.";
	}
	close;

L_Falcon:
	if(BaseJob != Job_Archer && BaseClass == Job_Archer && checkfalcon() == 0) goto L_Falc;
	if(getskilllv(127)<1) goto L_Need_Skill2;

	if (#gl_lang == 1) {
		mes "[Універсальна оренда]";
		mes "Вибач, " + strcharinfo(0) + ", але я здаю соколів лише Хантерам та Снайперам, які вміють з ними поводитися.";
	}
	if (#gl_lang < 1) {
		mes "[Universal Rental Npc]";
		mes "Sorry " + strcharinfo(0) + ", but I only rent falcons to Hunters and Snipers who the ability to handle 'em.";
	}
	close;

L_Falc:
	setfalcon;
	goto L_Quit2;

L_Need_Skill2:
	if (#gl_lang == 1) {
		mes "[Універсальна оренда]";
		mes "Вибач, у тебе немає необхідної навички, щоб володіти Соколом.";
	}
	if (#gl_lang < 1) {
		mes "[Universal Rental Npc]";
		mes "Sorry you don't have the required skill to own a Falcon.";
	}
	close;

L_Peco:
	if (BaseJob != Job_Swordman && BaseClass == Job_Swordman && checkriding() == 0) goto L_Peco_Ok;
	if(getskilllv(63)<1) goto L_Need_Skill3;

	if (#gl_lang == 1) {
		mes "[Універсальна оренда]";
		mes "Вибач, " + strcharinfo(0) + ", але я здаю Пекопеко лише Лицарям та Хрестоносцям, які вміють ними керувати.";
	}
	if (#gl_lang < 1) {
		mes "[Universal Rental Npc]";
		mes "Sorry " + strcharinfo(0) + ", but I only rent Pecopecos to Knights and Crusaders who have the ability to handle 'em.";
	}
	close;

L_Peco_Ok:
	setriding;
	goto L_Quit2;

L_Need_Skill3:
	if (#gl_lang == 1) {
		mes "[Універсальна оренда]";
		mes "Вибач, у тебе немає необхідної навички для їзди на Пекопеко.";
	}
	if (#gl_lang < 1) {
		mes "[Universal Rental Npc]";
		mes "Sorry you don't have the required skill to ride a Peco Peco.";
	}
	close;

L_Quit:
	if (#gl_lang == 1) {
		mes "[Універсальна оренда]";
		mes strcharinfo(0) + ", будь ласка, повертайся, коли будеш готовий щось орендувати.";
	}
	if (#gl_lang < 1) {
		mes "[Universal Rental Npc]";
		mes strcharinfo(0) + ", please come back when you are ready to rent something.";
	}
	close;

L_Quit2:
	if (#gl_lang == 1) {
		mes "[Універсальна оренда]";
		mes strcharinfo(0) + ", будь ласка, приходь ще, коли захочеш чогось іншого...";
	}
	if (#gl_lang < 1) {
		mes "[Universal Rental Npc]";
		mes strcharinfo(0) + ", please come again when you want another...";
	}
	close;
}
