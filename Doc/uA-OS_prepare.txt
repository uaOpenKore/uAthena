// Installation CentOS 7 x32 with building custom vanilla kernel 5.4+
// DOC updated 2024/10/29
// Recommended: CPU: 2, RAM: 3Gb, HDD: 40Gb, without swap part (file 1G)
// Minimum: 2/2/15/2
// Max: 4/3/60/1
// qemu: generic, partitions: standart ext4

1. yum install dnf* mc -y
1.1. dnf remove plymouth* abrt* dracut-config-rescue
1.2. rm -f /boot/*rescue*
1.3. ~ you might have to remove the rescue entry from /boot/loader/entries
2. systemctl disable firewalld
3. /etc/selinux/config -> disabled
-4. dnf install https://dl.fedoraproject.org/pub/epel/epel-release-latest-7.noarch.rpm (not supported :-( )
5. dnf install https://repo.mysql.com//mysql57-community-release-el7-11.noarch.rpm
7. enable repos -> /etc/yum.repos.d (enable mysql to 5.5)
7.1. dnf update -y
7.2. dnf install qemu-guest-agent
7.3. systemctl enable qemu-guest-agent

8. reboot

9. dnf groupinstall "Development Tools"
9.1. dnf install mysql-community-devel mysql-community-server gcc make perl kernel-devel kernel-headers bzip2 pcre-devel git zlib-devel glibc-devel ncurses-devel hmaccalc binutils-devel elfutils-libelf-devel bc openssl-devel rpm-build

10.1. dd if=/dev/zero of=/swapfile bs=1MB count=1024
10.2. chmod 600 /swapfile
10.3. mkswap /swapfile
10.4. swapon /swapfile
10.5. /etc/fstab -> /swapfile swap swap defaults 0 0
10.6. /etc/sysctl.conf - > vm.swappiness = 5
10.7. reboot

11.1. dnf install rpmdevtools dnf-auto* yum-* attr texinfo libacl-devel libncurses-dev

11.2. ~# rpmdev-setuptree
~extended:
11.3. dnf install bpftool kernel-cross-headers kernel-modules-extra kernel-modules-internal perf perl-generators perl-interpreter python3-perf mock python2-koj* python3-devel
11.4. useradd mockbuild
11.5. useradd mock
11.6. usermod -G mock mockbuild
